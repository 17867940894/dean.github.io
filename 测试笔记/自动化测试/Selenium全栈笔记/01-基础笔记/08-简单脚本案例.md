## 案例一：百度搜索并打印结果标题

**知识点**：环境搭建、定位、点击、等待、获取文本

```python
"""  
案例一：百度搜索并打印前 5 条结果标题  
"""  
from selenium import webdriver  
from selenium.webdriver.support.wait import WebDriverWait  
from selenium.webdriver.support import expected_conditions as EC  
from selenium.webdriver.common.by import By  
from time import sleep  
  
BAIDU_SEARCH_ID = (By.ID, "chat-textarea")  
BAIDU_SEARCH_BUTTON = (By.ID, "chat-submit-button")  
BAIDU_RESULT = (By.ID, "wrapper_wrapper")  
# 1. 初始化浏览器  
wd = webdriver.Chrome()  
wd.maximize_window()  
# 2. 打开百度  
wd.get("https://www.baidu.com")  
# 3. 显式等待搜索框可见并输入  
search_input = WebDriverWait(wd, 10).until(  
    EC.visibility_of_element_located(BAIDU_SEARCH_ID)  
)  
search_input.send_keys("Hello World")  
# 4. 显式等待按钮可点击  
search_button = WebDriverWait(wd, 10).until(  
    EC.element_to_be_clickable(BAIDU_SEARCH_BUTTON)  
)  
search_button.click()  
# 5. 等待结果加载  
WebDriverWait(wd, 10).until(  
    EC.visibility_of_element_located(BAIDU_RESULT)  
)  
print("搜索结果加载完成")  
# 6. 获取页面标题  
print(wd.title)  
sleep(5)
wd.quit()
```

## 案例五：使用 Cookie 跳过登录

**知识点**：Cookie 操作、复用登录态

```python
"""
案例五：用已保存的 Cookie 跳过登录直接访问受保护页面 
"""
from selenium import webdriver
from selenium.webdriver.common.by import By
import json

wd = webdriver.Chrome()

# 必须先打开同域页面，才能注入 Cookie
wd.get('https://www.example.com')

# 加载之前保存的 Cookie（实际从文件读取）
with open('cookies.json', 'r', encoding='utf-8') as f:
    cookies = json.load(f)

for cookie in cookies:
    # 部分 Cookie 字段 Selenium 不接受，需要清理
    for k in ['expiry', 'storeId']:
        cookie.pop(k, None)
    wd.add_cookie(cookie)

# 刷新，Cookie 生效
wd.refresh()

# 现在应该已是登录状态，直接访问受保护页面
wd.get('https://www.example.com/user/profile')
print('当前页面标题:', wd.title)

wd.quit()
```

## 案例六：异常捕获与重试

**知识点**：异常捕获、健壮性设计

```python
"""
案例六：演示基础异常捕获，提升脚本健壮性
"""
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.common.exceptions import (
    NoSuchElementException,
    TimeoutException,
    ElementNotInteractableException
)
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
import time

def safe_click(driver, locator, timeout=10, retry=2):
    """安全点击：失败自动重试"""
    for attempt in range(retry + 1):
        try:
            el = WebDriverWait(driver, timeout).until(
                EC.element_to_be_clickable(locator)
            )
            el.click()
            return True
        except (TimeoutException, ElementNotInteractableException) as e:
            print(f'第 {attempt+1} 次点击失败: {e}')
            if attempt < retry:
                time.sleep(1)
            else:
                raise
    return False

wd = webdriver.Chrome()
wd.get('https://www.baidu.com')

try:
    safe_click(wd, (By.ID, 'kw'))
    wd.find_element(By.ID, 'kw').send_keys('Selenium')
    safe_click(wd, (By.ID, 'su'))
except NoSuchElementException as e:
    print('元素找不到:', e)
except TimeoutException as e:
    print('超时:', e)
finally:
    wd.quit()
```
