# PO 模式（Page Object）

> [!info] 本节目标
> 理解 PO 模式的核心思想与分层结构，掌握 BasePage + 业务 Page 的标准写法，把"页面元素"与"测试逻辑"彻底解耦。

相关笔记：[[01-显式等待封装]]、[[10-异常捕获]]、实战项目 `pages/`

## 一、为什么需要 PO 模式

### 1.1 脚本式写法的问题

```python
# 脚本式：定位 + 操作 + 断言混在一起
def test_login():
    driver.get('https://www.example.com/login')
    driver.find_element(By.ID, 'username').send_keys('admin')
    driver.find_element(By.ID, 'password').send_keys('123456')
    driver.find_element(By.ID, 'login').click()
    assert '首页' in driver.title
```

问题：

- **元素定位散落各处**：UI 一改，所有用例都要找着改
- **无法复用**：登录逻辑在其他用例里要重写一遍
- **难以维护**：用例数到 50+ 时就成了噩梦

### 1.2 PO 模式的核心思想

> **Page Object 模式：把每个页面抽象成一个类，页面的元素定位、操作方法都封装在类里，测试用例只调用方法、做断言。**

UI 变了，只改页面对象类，测试用例不动。

## 二、PO 分层结构

```
testcases/          测试用例层：只关心业务流程和断言
   ↑ 调用
pages/              页面对象层：封装元素定位 + 页面操作
   ↑ 继承
pages/base_page.py  基类：所有页面的公共方法（点击、输入、等待、截图）
   ↑ 依赖
utils/              工具层：浏览器、等待、日志、截图、数据读取
```

## 三、BasePage 基类

```python
"""
BasePage：所有页面对象的基类
位置：pages/base_page.py
"""
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from utils.wait_helper import WaitHelper
from utils.logger import logger
from utils.screenshot import screenshot_on_failure
import time


class BasePage:
    """页面基类，封装所有页面共用的操作"""

    def __init__(self, driver):
        self.driver = driver
        self.wait = WaitHelper(driver, timeout=10)

    # ---------- 基础操作 ----------

    def open(self, url):
        """打开页面"""
        self.driver.get(url)
        logger.info(f'打开页面: {url}')
        return self

    def find(self, locator, timeout=None):
        """查找可见元素"""
        return self.wait.element_visible(locator, timeout)

    def finds(self, locator, timeout=None):
        """查找所有匹配元素"""
        return self.wait.elements_present(locator, timeout)

    def click(self, locator, timeout=None):
        """点击元素"""
        el = self.wait.element_clickable(locator, timeout)
        el.click()
        logger.info(f'点击元素: {locator}')
        return self

    def type(self, locator, text, clear_first=True, timeout=None):
        """输入文本"""
        el = self.wait.element_visible(locator, timeout)
        if clear_first:
            el.clear()
        el.send_keys(text)
        logger.info(f'输入文本到 {locator}: {text}')
        return self

    def get_text(self, locator, timeout=None):
        """获取元素文本"""
        return self.find(locator, timeout).text

    def get_attribute(self, locator, attr, timeout=None):
        """获取元素属性"""
        return self.find(locator, timeout).get_attribute(attr)

    # ---------- 等待相关 ----------

    def wait_url_contains(self, text, timeout=None):
        self.wait.url_contains(text, timeout)
        return self

    def wait_title_contains(self, text, timeout=None):
        self.wait.title_contains(text, timeout)
        return self

    # ---------- 切换相关 ----------

    def switch_to_frame(self, locator, timeout=None):
        self.wait.frame_and_switch(locator, timeout)
        return self

    def switch_to_default(self):
        self.driver.switch_to.default_content()
        return self

    def switch_to_new_window(self, timeout=None):
        self.wait.new_window(timeout)
        return self

    # ---------- 断言辅助 ----------

    def is_element_visible(self, locator, timeout=2):
        """快速判断元素是否可见"""
        return self.wait.is_element_visible(locator, timeout)

    # ---------- 截图 ----------

    def screenshot(self, name=None):
        """截图"""
        name = name or f'screenshot_{int(time.time())}'
        path = screenshot_on_failure(self.driver, name)
        logger.info(f'截图已保存: {path}')
        return path
```

## 四、业务页面对象

```python
"""
登录页对象
位置：pages/login_page.py
"""
from selenium.webdriver.common.by import By
from pages.base_page import BasePage


class LoginPage(BasePage):
    """登录页：封装登录相关的元素和操作"""

    # ============ 元素定位器 ============
    # 集中管理，UI 变了只改这里
    URL = 'https://www.example.com/login'
    username_input = (By.ID, 'username')
    password_input = (By.ID, 'password')
    login_btn = (By.ID, 'login')
    error_msg = (By.CSS_SELECTOR, '.error-message')
    remember_checkbox = (By.ID, 'remember')

    # ============ 页面操作 ============

    def load(self):
        """打开登录页"""
        self.open(self.URL)
        return self

    def login(self, username, password, remember=False):
        """登录操作"""
        self.type(self.username_input, username)
        self.type(self.password_input, password)
        if remember and not self.driver.find_element(*self.remember_checkbox).is_selected():
            self.click(self.remember_checkbox)
        self.click(self.login_btn)
        return self

    def get_error_message(self):
        """获取错误提示"""
        return self.get_text(self.error_msg, timeout=3)

    def is_login_button_enabled(self):
        """判断登录按钮是否可点"""
        return self.find(self.login_btn).is_enabled()
```

```python
"""
首页对象
位置：pages/home_page.py
"""
from selenium.webdriver.common.by import By
from pages.base_page import BasePage


class HomePage(BasePage):
    """首页：登录后进入的页面"""

    URL = 'https://www.example.com/home'
    welcome_text = (By.CSS_SELECTOR, '.welcome')
    logout_btn = (By.ID, 'logout')
    user_menu = (By.CSS_SELECTOR, '.user-menu')

    def is_loaded(self):
        """判断首页是否加载完成"""
        return self.is_element_visible(self.welcome_text, timeout=5)

    def get_welcome_text(self):
        return self.get_text(self.welcome_text)

    def logout(self):
        # 悬停用户菜单后点击退出
        from selenium.webdriver import ActionChains
        actions = ActionChains(self.driver)
        actions.move_to_element(self.find(self.user_menu)).perform()
        self.click(self.logout_btn)
        return self
```

## 五、测试用例

```python
"""
登录测试用例
位置：testcases/test_login.py
"""
import pytest
from pages.login_page import LoginPage
from pages.home_page import HomePage


class TestLogin:

    def setup_method(self):
        self.login_page = LoginPage(self.driver)

    def test_login_success(self):
        """正常登录"""
        self.login_page.load().login('admin', 'Admin@123')
        home = HomePage(self.driver)
        assert home.is_loaded()
        assert '欢迎' in home.get_welcome_text()

    def test_login_wrong_password(self):
        """密码错误"""
        self.login_page.load().login('admin', 'wrongpass')
        assert '密码错误' in self.login_page.get_error_message()

    def test_login_empty_username(self):
        """用户名为空"""
        self.login_page.load().login('', '123456')
        assert self.login_page.is_login_button_enabled() is False
```

> [!note] setup_method 中的 self.driver
> `self.driver` 通常由 pytest fixture 注入，详见实战项目 `conftest.py`。

## 六、PO 模式的好处

| 维度 | 脚本式 | PO 模式 |
| --- | --- | --- |
| 元素定位 | 散落各处 | 集中在页面类顶部 |
| UI 变更 | 改所有用例 | 只改页面类 |
| 复用 | 无 | 登录逻辑可被所有用例调用 |
| 可读性 | 流程+定位+断言混杂 | 用例只剩业务流程 |
| 团队协作 | 冲突多 | 页面/用例分层开发 |

## 七、PO 模式最佳实践

1. **不要在 Page 里写断言**：断言属于测试用例，Page 只提供数据和操作
2. **方法返回 self**：支持链式调用 `page.load().login().logout()`
3. **定位器集中声明**：放在类顶部，命名清晰
4. **页面间跳转返回目标页对象**：`login()` 成功后 `return HomePage(self.driver)`
5. **公共方法提到 BasePage**：避免重复
6. **页面粒度适中**：一个页面一个类，不要把整站塞一个类
7. **配合 fixture 管理 driver**：driver 生命周期由 pytest fixture 控制

## 八、页面跳转返回目标页对象

```python
class LoginPage(BasePage):
    def login_success(self, u, p):
        self.login(u, p)
        # 登录成功后跳转到首页，直接返回首页对象
        return HomePage(self.driver)
```

用例中：

```python
def test_login(self):
    home = self.login_page.load().login_success('admin', 'Admin@123')
    assert home.is_loaded()
```
