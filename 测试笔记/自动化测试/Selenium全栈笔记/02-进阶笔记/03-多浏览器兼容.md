# 多浏览器兼容

> [!info] 本节目标
> 封装统一的浏览器工厂，支持通过配置切换 Chrome / Edge / Firefox / Headless，并处理好各浏览器的差异（驱动、选项、中文路径）。

相关笔记：[[02-浏览器驱动]]、[[05-截图日志]]、实战项目 `utils/browser.py`

## 一、为什么需要多浏览器兼容

- **兼容性测试**：同一套用例要在 Chrome、Edge、Firefox 上都跑一遍
- **环境差异**：开发用 Chrome，CI 服务器用 Headless
- **客户环境**：金融、政企客户常用 Edge / 国产浏览器

## 二、各浏览器驱动差异速查

| 浏览器 | Driver | 驱动类 | Options 类 |
| --- | --- | --- | --- |
| Chrome | chromedriver | `webdriver.Chrome` | `ChromeOptions` |
| Edge | msedgedriver | `webdriver.Edge` | `EdgeOptions` |
| Firefox | geckodriver | `webdriver.Firefox` | `FirefoxOptions` |
| Safari | safaridriver | `webdriver.Safari` | `SafariOptions` |

## 三、通用浏览器工厂封装

```python
"""
浏览器工厂：统一创建各类型浏览器实例
位置：utils/browser.py
"""
import os
from selenium import webdriver
from selenium.webdriver.chrome.service import Service as ChromeService
from selenium.webdriver.edge.service import Service as EdgeService
from selenium.webdriver.firefox.service import Service as FirefoxService
from utils.logger import logger


class BrowserFactory:
    """根据浏览器名称创建 WebDriver 实例"""

    # 可选镜像源（国内网络问题）
    CHROME_MIRROR = 'https://cdn.npmmirror.com/binaries/chrome-for-testing'

    @classmethod
    def create(cls, browser='chrome', headless=False, maximize=True,
               driver_path=None, window_size=None, extra_options=None):
        """
        :param browser: chrome / edge / firefox
        :param headless: 是否无头模式
        :param maximize: 是否最大化窗口
        :param driver_path: 手动指定驱动路径（可选）
        :param window_size: (width, height) 窗口大小
        :param extra_options: 额外 options 列表
        :return: WebDriver 实例
        """
        browser = browser.lower().strip()
        logger.info(f'创建浏览器: {browser}, headless={headless}')

        if browser == 'chrome':
            driver = cls._create_chrome(headless, driver_path, extra_options)
        elif browser == 'edge':
            driver = cls._create_edge(headless, driver_path, extra_options)
        elif browser == 'firefox':
            driver = cls._create_firefox(headless, driver_path, extra_options)
        else:
            raise ValueError(f'不支持的浏览器类型: {browser}')

        # 窗口设置
        if maximize:
            driver.maximize_window()
        elif window_size:
            driver.set_window_size(*window_size)

        # 页面加载超时
        driver.set_page_load_timeout(30)

        logger.info(f'浏览器创建成功: {browser}')
        return driver

    @classmethod
    def _create_chrome(cls, headless, driver_path, extra_options):
        # 国内网络：指定驱动镜像源
        os.environ.setdefault('SE_DRIVER_MIRROR_URL', cls.CHROME_MIRROR)

        options = webdriver.ChromeOptions()
        # 关闭 chromedriver 日志
        options.add_experimental_option('excludeSwitches', ['enable-logging'])
        # 禁用"Chrome 正在被自动测试软件控制"提示（旧版）
        options.add_experimental_option('useAutomationExtension', False)
        # 去除自动化特征（防检测）
        options.add_argument('--disable-blink-features=AutomationControlled')

        if headless:
            options.add_argument('--headless=new')  # Selenium 4 新写法
            options.add_argument('--disable-gpu')
            options.add_argument('--no-sandbox')
            options.add_argument('--disable-dev-shm-usage')

        if extra_options:
            for opt in extra_options:
                options.add_argument(opt)

        service = ChromeService(executable_path=driver_path) if driver_path else ChromeService()
        return webdriver.Chrome(service=service, options=options)

    @classmethod
    def _create_edge(cls, headless, driver_path, extra_options):
        options = webdriver.EdgeOptions()
        options.add_experimental_option('excludeSwitches', ['enable-logging'])

        if headless:
            options.add_argument('--headless=new')
            options.add_argument('--disable-gpu')

        if extra_options:
            for opt in extra_options:
                options.add_argument(opt)

        service = EdgeService(executable_path=driver_path) if driver_path else EdgeService()
        return webdriver.Edge(service=service, options=options)

    @classmethod
    def _create_firefox(cls, headless, driver_path, extra_options):
        options = webdriver.FirefoxOptions()

        if headless:
            options.add_argument('--headless')

        if extra_options:
            for opt in extra_options:
                options.add_argument(opt)

        service = FirefoxService(executable_path=driver_path) if driver_path else FirefoxService()
        return webdriver.Firefox(service=service, options=options)
```

## 四、通过配置文件切换浏览器

```yaml
# config/config.yaml
browser:
  name: chrome          # chrome / edge / firefox
  headless: false
  maximize: true
  driver_path: null     # 手动指定路径，留空让 Selenium Manager 自动管理
  window_size: null     # [1280, 800] 或 null
```

```python
# utils/config_loader.py
import yaml
import os

def load_config():
    config_path = os.path.join(os.path.dirname(__file__), '..', 'config', 'config.yaml')
    with open(config_path, 'r', encoding='utf-8') as f:
        return yaml.safe_load(f)
```

```python
# conftest.py 中使用
import pytest
from utils.browser import BrowserFactory
from utils.config_loader import load_config

config = load_config()

@pytest.fixture(scope='session')
def driver():
    browser_cfg = config['browser']
    driver = BrowserFactory.create(
        browser=browser_cfg['name'],
        headless=browser_cfg.get('headless', False),
        maximize=browser_cfg.get('maximize', True),
        driver_path=browser_cfg.get('driver_path'),
        window_size=browser_cfg.get('window_size'),
    )
    yield driver
    driver.quit()
```

## 五、多浏览器并行测试

用 pytest 的参数化实现"同一套用例在多个浏览器跑"：

```python
# conftest.py
import pytest
from utils.browser import BrowserFactory

BROWSERS = ['chrome', 'edge', 'firefox']

def pytest_addoption(parser):
    parser.addoption('--browser', action='append', default=[], help='浏览器列表')

def pytest_generate_tests(metafunc):
    if 'browser_name' in metafunc.fixturenames:
        browsers = metafunc.config.getoption('--browser') or BROWSERS
        metafunc.parametrize('browser_name', browsers, scope='session')

@pytest.fixture(scope='session')
def driver(browser_name):
    d = BrowserFactory.create(browser=browser_name, headless=False)
    yield d
    d.quit()
```

```bash
# 命令行指定要跑的浏览器
pytest testcases/ --browser chrome --browser edge
```

## 六、各浏览器差异与坑

### 6.1 Firefox 中文路径

Firefox 对中文路径敏感，截图、下载路径含中文可能出问题。建议：

```python
options = webdriver.FirefoxOptions()
# 截图、下载用英文路径
```

### 6.2 Chrome 无头模式渲染差异

`--headless`（旧）和 `--headless=new`（新）渲染有差异。新版 `--headless=new` 渲染更接近有头模式，**推荐用 new**。

### 6.3 Edge 与 Chrome API 几乎一致

Edge 基于 Chromium，`ChromeOptions` 的大部分参数在 `EdgeOptions` 上也能用。

### 6.4 国产浏览器（360 / QQ / 搜狗）

基于 Chromium 内核，可以用 Chrome 驱动，但需指定浏览器二进制路径：

```python
options = webdriver.ChromeOptions()
options.binary_location = r'C:\Users\xxx\AppData\Roaming\360se6\Application\360se.exe'
driver = webdriver.Chrome(options=options)
```

## 七、防自动化检测

部分网站会检测 Selenium 特征（`navigator.webdriver=true`），可通过以下方式降低检测：

```python
options = webdriver.ChromeOptions()
# 去除自动化特征
options.add_argument('--disable-blink-features=AutomationControlled')
options.add_experimental_option('excludeSwitches', ['enable-logging'])
options.add_experimental_option('useAutomationExtension', False)

driver = webdriver.Chrome(options=options)

# 进一步修改 webdriver 标志
driver.execute_cdp_cmd('Page.addScriptToEvaluateOnNewDocument', {
    'source': 'Object.defineProperty(navigator, "webdriver", {get: () => undefined})'
})
```

> [!warning] 注意
> 防检测手段只能"降低"被识别概率，强反爬站点（如淘宝、12306）仍可能识别。商业爬取请遵守目标站点 robots 协议与法律法规。
