## 一、日志工具封装

### 1.1 为什么不用 print

- `print` 无法分级（DEBUG / INFO / WARNING / ERROR）
- `print` 无法同时输出到控制台和文件
- `print` 无时间戳、无模块名
- 生产代码必须用 `logging`

### 1.2 Logger 封装

```python
"""
日志工具类
位置：utils/logger.py
"""
import logging
import os
from datetime import datetime


def setup_logger(log_dir='logs', name='selenium', level=logging.INFO,
                 console=True, file_flag=True):
    """
    初始化并返回 logger
    :param log_dir: 日志目录
    :param name: logger 名称
    :param level: 日志级别
    :param console: 是否输出到控制台
    :param file_flag: 是否输出到文件
    :return: logging.Logger
    """
    logger = logging.getLogger(name)

    # 避免重复添加 handler（多次调用时）
    if logger.handlers:
        return logger

    logger.setLevel(level)
    formatter = logging.Formatter(
        '%(asctime)s [%(levelname)s] [%(filename)s:%(lineno)d] %(message)s',
        datefmt='%Y-%m-%d %H:%M:%S'
    )

    # 控制台输出
    if console:
        ch = logging.StreamHandler()
        ch.setLevel(level)
        ch.setFormatter(formatter)
        logger.addHandler(ch)

    # 文件输出（按日期分文件）
    if file_flag:
        os.makedirs(log_dir, exist_ok=True)
        log_file = os.path.join(log_dir, f'{name}_{datetime.now():%Y%m%d}.log')
        fh = logging.FileHandler(log_file, encoding='utf-8')
        fh.setLevel(level)
        fh.setFormatter(formatter)
        logger.addHandler(fh)

    return logger


# 模块级单例：导入即用
logger = setup_logger()
```

### 1.3 使用示例

```python
from utils.logger import logger

logger.debug('调试信息')
logger.info('正常信息：打开页面 https://...')
logger.warning('警告：元素加载较慢')
logger.error('错误：登录失败')
logger.exception('带堆栈的异常')  # 自动附带 traceback
```

输出示例：

```
2026-07-17 10:20:35 [INFO] [login_page.py:42] 正常信息：打开页面 https://...
2026-07-17 10:20:36 [ERROR] [login_page.py:58] 错误：登录失败
```

### 1.4 日志级别使用规范

| 级别 | 何时使用 |
| --- | --- |
| DEBUG | 详细调试信息（开发阶段用） |
| INFO | 关键业务节点（打开页面、点击按钮、登录成功） |
| WARNING | 非预期但可继续（元素加载慢、重试中） |
| ERROR | 业务失败（登录失败、断言失败） |
| CRITICAL | 系统级故障（浏览器崩溃、无法启动） |

## 二、截图工具封装

### 2.1 基础截图 `save_screenshot`

```python
from selenium import webdriver
import os
from datetime import datetime

def screenshot(driver, name='screenshot', save_dir='screenshots'):
    """
    普通截图（只截可视区域）
    :return: 截图文件路径
    """
    os.makedirs(save_dir, exist_ok=True)
    filename = f'{name}_{datetime.now():%Y%m%d_%H%M%S}.png'
    filepath = os.path.join(save_dir, filename)
    driver.save_screenshot(filepath)
    return filepath
```

### 2.2 全页面截图（滚动拼接）

普通 `save_screenshot` 只截可视区域，长页面需要滚动拼接：

```python
def full_page_screenshot(driver, name='fullpage', save_dir='screenshots'):
    """全页面截图（Chrome/Edge 用 CDP 命令）"""
    os.makedirs(save_dir, exist_ok=True)
    filepath = os.path.join(save_dir, f'{name}_{datetime.now():%Y%m%d_%H%M%S}.png')

    # Chrome / Edge（基于 CDP，最准）
    try:
        # 获取页面总高度
        body = driver.execute_script('return document.body.scrollHeight')
        viewport = driver.execute_script('return window.innerHeight')
        driver.set_window_size(driver.execute_script('return window.outerWidth'),
                               body + driver.execute_script('return window.outerHeight - window.innerHeight'))
        driver.save_screenshot(filepath)
        return filepath
    except Exception as e:
        # 降级为普通截图
        driver.save_screenshot(filepath)
        return filepath
```

### 2.3 元素截图 `element.screenshot()`

```python
def element_screenshot(driver, locator, name='element', save_dir='screenshots'):
    """对单个元素截图（高清）"""
```

### 2.4 失败自动截图（pytest 集成）

用 pytest 的 hook 在用例失败时自动截图：

```python
"""
位置：conftest.py
"""
import pytest
import os
from datetime import datetime
from utils.logger import logger


@pytest.hookimpl(hookwrapper=True, tryfirst=True)
def pytest_runtest_makereport(item, call):
    """用例执行结果生成时插入截图逻辑"""
    outcome = yield
    report = outcome.get_result()

    if report.when == 'call' and report.failed:
        # 从 item 中拿到 driver（fixture 注入）
        driver = item.funcargs.get('driver')
        if driver:
            try:
                screenshot_dir = 'screenshots'
                os.makedirs(screenshot_dir, exist_ok=True)
                name = f'FAIL_{item.name}_{datetime.now():%Y%m%d_%H%M%S}'
                filepath = os.path.join(screenshot_dir, f'{name}.png')
                driver.save_screenshot(filepath)
                logger.error(f'用例失败，截图: {filepath}')

                # 把截图附加到 pytest-html 报告（如果装了）
                if pytest_html := item.config.pluginmanager.getplugin('html'):
                    extra = getattr(report, 'extra', [])
                    extra.append(pytest_html.extras.image(filepath))
                    report.extra = extra
            except Exception as e:
                logger.error(f'截图失败: {e}')
```

### 2.5 完整 ScreenshotHelper 工具类

```python
"""
截图工具类
位置：utils/screenshot.py
"""
import os
from datetime import datetime
from utils.logger import logger


class ScreenshotHelper:

    def __init__(self, driver, save_dir='screenshots'):
        self.driver = driver
        self.save_dir = save_dir
        os.makedirs(save_dir, exist_ok=True)

    def take(self, name='screenshot'):
        """普通截图"""
        filepath = os.path.join(
            self.save_dir,
            f'{name}_{datetime.now():%Y%m%d_%H%M%S}.png'
        )
        self.driver.save_screenshot(filepath)
        logger.info(f'截图: {filepath}')
        return filepath

    def take_element(self, locator, name='element'):
        """元素截图"""
        filepath = os.path.join(
            self.save_dir,
            f'{name}_{datetime.now():%Y%m%d_%H%M%S}.png'
        )
        element = self.driver.find_element(*locator)
        element.screenshot(filepath)
        logger.info(f'元素截图: {filepath}')
        return filepath

    def take_full_page(self, name='fullpage'):
        """全页面截图"""
        filepath = os.path.join(
            self.save_dir,
            f'{name}_{datetime.now():%Y%m%d_%H%M%S}.png'
        )
        # 利用 CDP 设置设备 metrics 模拟整页
        try:
            width = self.driver.execute_script('return document.body.scrollWidth')
            height = self.driver.execute_script('return document.body.scrollHeight')
            self.driver.execute_cdp_cmd('Emulation.setDeviceMetricsOverride', {
                'width': width,
                'height': height,
                'deviceScaleFactor': 1,
                'mobile': False
            })
            self.driver.save_screenshot(filepath)
            # 恢复
            self.driver.execute_cdp_cmd('Emulation.clearDeviceMetricsOverride', {})
        except Exception:
            self.driver.save_screenshot(filepath)
        logger.info(f'全页面截图: {filepath}')
        return filepath


# 便捷函数
def screenshot_on_failure(driver, name):
    """失败时调用的便捷函数"""
    helper = ScreenshotHelper(driver)
    return helper.take(name)
```

## 三、在 BasePage 中集成

```python
class BasePage:
    def __init__(self, driver):
        self.driver = driver
        self.sh = ScreenshotHelper(driver)

    def screenshot(self, name=None):
        return self.sh.take(name or self.__class__.__name__)
```

## 四、日志与截图的关系

| 维度 | 日志 | 截图 |
| --- | --- | --- |
| 作用 | 记录操作过程 | 保留失败现场 |
| 时机 | 每个关键操作都记 | 用例失败时自动截 |
| 体积 | 小（文本） | 大（PNG） |
| 排查价值 | 看流程 | 看页面状态 |

> [!tip] 最佳实践
> - 日志：INFO 级记录"打开页面、点击、输入、断言结果"
> - 截图：用例失败自动截 + 关键步骤手动截
> - 报告：用 `pytest-html` 把截图嵌入 HTML 报告，详见实战项目 `run.py`
