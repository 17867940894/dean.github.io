# 显式等待封装

> [!info] 本节目标
> 把 `WebDriverWait` 封装成可复用的 `WaitHelper` 工具类，覆盖元素可见、可点击、消失、文本出现、自定义条件等场景，统一项目等待规范。

相关笔记：[[05-等待机制]]、[[02-PO模式]]、[[10-异常捕获]]

## 一、为什么需要封装

裸用 `WebDriverWait` 的问题：

```python
# 每次都写一长串，重复且易错
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.webdriver.common.by import By

element = WebDriverWait(driver, 10, 0.5).until(
    EC.visibility_of_element_located((By.ID, 'kw'))
)
```

封装后：

```python
# 一行搞定
element = wait_helper.element_visible((By.ID, 'kw'))
```

## 二、WaitHelper 完整封装

```python
"""
显式等待工具类
位置：utils/wait_helper.py
"""
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.common.exceptions import TimeoutException, NoSuchElementException


class WaitHelper:
    """统一管理项目中的显式等待"""

    def __init__(self, driver, timeout=10, poll_frequency=0.5):
        """
        :param driver: WebDriver 实例
        :param timeout: 默认超时时间（秒）
        :param poll_frequency: 轮询间隔（秒）
        """
        self.driver = driver
        self.timeout = timeout
        self.poll_frequency = poll_frequency

    def _wait(self, timeout=None):
        """构造 WebDriverWait 对象"""
        return WebDriverWait(
            self.driver,
            timeout if timeout is not None else self.timeout,
            self.poll_frequency
        )

    # ---------- 元素相关 ----------

    def element_present(self, locator, timeout=None):
        """等待元素出现在 DOM 中（不一定可见）"""
        return self._wait(timeout).until(
            EC.presence_of_element_located(locator),
            message=f'元素未出现在 DOM 中: {locator}'
        )

    def element_visible(self, locator, timeout=None):
        """等待元素可见"""
        return self._wait(timeout).until(
            EC.visibility_of_element_located(locator),
            message=f'元素不可见: {locator}'
        )

    def element_clickable(self, locator, timeout=None):
        """等待元素可点击（可见且 enabled）"""
        return self._wait(timeout).until(
            EC.element_to_be_clickable(locator),
            message=f'元素不可点击: {locator}'
        )

    def elements_present(self, locator, timeout=None):
        """等待至少一个匹配元素出现在 DOM 中，返回列表"""
        return self._wait(timeout).until(
            EC.presence_of_all_elements_located(locator),
            message=f'未找到任何匹配元素: {locator}'
        )

    def element_invisible(self, locator, timeout=None):
        """等待元素消失（不可见或从 DOM 移除）"""
        return self._wait(timeout).until(
            EC.invisibility_of_element_located(locator),
            message=f'元素未消失: {locator}'
        )

    def element_stale(self, element, timeout=None):
        """等待元素已从 DOM 移除（引用失效）"""
        return self._wait(timeout).until(
            EC.staleness_of(element),
            message=f'元素未失效: {element}'
        )

    # ---------- 文本/属性相关 ----------

    def text_in_element(self, locator, text, timeout=None):
        """等待元素中包含指定文本"""
        return self._wait(timeout).until(
            EC.text_to_be_present_in_element(locator, text),
            message=f'元素中未出现文本 "{text}": {locator}'
        )

    def text_in_element_value(self, locator, text, timeout=None):
        """等待元素 value 属性包含指定文本"""
        return self._wait(timeout).until(
            EC.text_to_be_present_in_element_value(locator, text),
            message=f'元素 value 中未出现文本 "{text}": {locator}'
        )

    # ---------- 页面相关 ----------

    def title_is(self, title, timeout=None):
        """等待页面标题等于指定值"""
        return self._wait(timeout).until(
            EC.title_is(title),
            message=f'页面标题不是 "{title}"，当前: {self.driver.title}'
        )

    def title_contains(self, text, timeout=None):
        """等待页面标题包含指定文本"""
        return self._wait(timeout).until(
            EC.title_contains(text),
            message=f'页面标题不含 "{text}"，当前: {self.driver.title}'
        )

    def url_contains(self, text, timeout=None):
        """等待 URL 包含指定文本"""
        return self._wait(timeout).until(
            EC.url_contains(text),
            message=f'URL 不含 "{text}"，当前: {self.driver.current_url}'
        )

    def url_to_be(self, url, timeout=None):
        """等待 URL 等于指定值"""
        return self._wait(timeout).until(
            EC.url_to_be(url),
            message=f'URL 不是 "{url}"，当前: {self.driver.current_url}'
        )

    # ---------- 弹窗/iframe/窗口 ----------

    def alert_present(self, timeout=None):
        """等待 alert 弹窗出现"""
        return self._wait(timeout).until(
            EC.alert_is_present(),
            message='alert 弹窗未出现'
        )

    def frame_and_switch(self, locator, timeout=None):
        """等待 iframe 可用并切换进去"""
        return self._wait(timeout).until(
            EC.frame_to_be_available_and_switch_to_it(locator),
            message=f'iframe 不可用: {locator}'
        )

    def new_window(self, timeout=None):
        """等待新窗口出现并切换过去"""
        old = set(self.driver.window_handles)
        self._wait(timeout).until(
            lambda d: len(d.window_handles) > len(old),
            message='未打开新窗口'
        )
        new = set(self.driver.window_handles) - old
        self.driver.switch_to.window(new.pop())

    # ---------- 自定义条件 ----------

    def until(self, condition, timeout=None, message=''):
        """通用：等待自定义条件（函数接收 driver，返回 True/元素）"""
        return self._wait(timeout).until(condition, message=message)

    def until_not(self, condition, timeout=None, message=''):
        """通用：等待条件变为 False"""
        return self._wait(timeout).until_not(condition, message=message)

    # ---------- 容错版 ----------

    def element_visible_or_none(self, locator, timeout=None):
        """等待元素可见，超时返回 None（不抛异常）"""
        try:
            return self.element_visible(locator, timeout)
        except TimeoutException:
            return None

    def is_element_visible(self, locator, timeout=2):
        """快速判断元素是否可见（短超时）"""
        return self.element_visible_or_none(locator, timeout) is not None
```

## 三、使用示例

### 3.1 在 PO 中使用

```python
from selenium.webdriver.common.by import By
from utils.wait_helper import WaitHelper

class LoginPage:
    def __init__(self, driver):
        self.driver = driver
        self.wait = WaitHelper(driver, timeout=10)

        # 定位器
        self.username_input = (By.ID, 'username')
        self.password_input = (By.ID, 'password')
        self.login_btn = (By.ID, 'login')

    def login(self, username, password):
        # 用封装的等待，代码清爽
        self.wait.element_visible(self.username_input).send_keys(username)
        self.wait.element_visible(self.password_input).send_keys(password)
        self.wait.element_clickable(self.login_btn).click()

        # 等待登录后页面跳转
        self.wait.url_contains('/home')
```

### 3.2 容错判断

```python
wait = WaitHelper(driver)

# 判断"退出登录"按钮是否存在（用户可能已登录或未登录）
if wait.is_element_visible((By.ID, 'logout'), timeout=3):
    print('已登录')
else:
    print('未登录，需要先登录')
```

### 3.3 自定义条件

```python
# 等待搜索结果数量大于 5 条
wait.until(
    lambda d: len(d.find_elements(By.CSS_SELECTOR, '.result')) > 5,
    message='搜索结果不足 5 条'
)
```

## 四、封装设计原则

1. **默认参数 + 可覆盖**：构造时设默认 timeout，调用时可单独覆盖
2. **错误信息友好**：超时时 message 直接告诉是哪个 locator 出问题
3. **返回 WebElement**：方便链式调用 `.send_keys()` `.click()`
4. **容错版本**：`xxx_or_none` / `is_xxx` 系列用于"判断而非等待"场景
5. **不混用隐式等待**：项目内统一只用显式等待（见 [[05-等待机制]] 警告）

## 五、与 BasePage 集成

`WaitHelper` 通常作为 `BasePage` 的成员，所有页面对象继承后直接使用：

```python
class BasePage:
    def __init__(self, driver):
        self.driver = driver
        self.wait = WaitHelper(driver, timeout=10)

    def find(self, locator):
        return self.wait.element_visible(locator)

    def click(self, locator):
        self.wait.element_clickable(locator).click()
```
